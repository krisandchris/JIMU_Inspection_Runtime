import argparse,datetime,json,os,pathlib,shlex,subprocess,sys,time
p=argparse.ArgumentParser();p.add_argument('--kind',choices=['test','static','dependency'],required=True);p.add_argument('command',nargs=argparse.REMAINDER);a=p.parse_args();cmd=a.command
if cmd and cmd[0]=='--':cmd=cmd[1:]
e=pathlib.Path('planning/implementation_evidence/WP-01A');started=datetime.datetime.now(datetime.timezone.utc).isoformat();start=time.monotonic()
env=os.environ.copy();env['UV_PYTHON_DOWNLOADS']='never';env['UV_DEFAULT_INDEX']='https://pypi.org/simple'
r=subprocess.run(cmd,capture_output=True,text=True,env=env);ended=datetime.datetime.now(datetime.timezone.utc).isoformat()
entry={'kind':a.kind,'command':shlex.join(cmd),'cwd':str(pathlib.Path.cwd()),'started_at':started,'finished_at':ended,'elapsed_seconds':round(time.monotonic()-start,3),'exit_code':r.returncode,'stdout':r.stdout,'stderr':r.stderr,'environment_overrides':{'UV_PYTHON_DOWNLOADS':'never','UV_DEFAULT_INDEX':'https://pypi.org/simple'}}
with (e/'command_results.jsonl').open('a') as f:f.write(json.dumps(entry)+'\n')
with (e/'test_commands.txt').open('a') as f:f.write(a.kind.upper()+' '+entry['command']+'\n')
target='test_results.txt' if a.kind=='test' else 'static_checks.txt'
with (e/target).open('a') as f:f.write('\n$ '+entry['command']+'\nkind='+a.kind+'\nstarted_at='+started+'\nfinished_at='+ended+'\nexit_code='+str(r.returncode)+'\n'+r.stdout+r.stderr+'\n')
print(entry['command']);print('exit_code='+str(r.returncode));print(r.stdout+r.stderr,end='')
sys.exit(r.returncode)
